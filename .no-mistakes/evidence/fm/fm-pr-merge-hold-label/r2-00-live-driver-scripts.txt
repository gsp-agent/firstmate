#!/usr/bin/env bash
# Build this round's disposable GitHub fixture: a marked lab home, a full
# base-commit tree, a clone of the private scratch repo, and fresh PRs.
set -euo pipefail
WT=/Users/dalecarman/.agent-scratch/nm-runs/a5e67f4ea77f/01M5Z5SSDSR36DJ1JGJNQZT1KG
W=/var/folders/2j/3tf17x5d7pj10gdjbp1ydpf40000gn/T/nm-hold-r2.5ycrT3
REPO=gsp-agent/fm-hold-label-proof
RUN=r2$(date -u +%H%M%S)
LAB="$W/lab"
export GH_TOKEN
GH_TOKEN=$(gh auth token --user gsp-agent)

"$WT/bin/fm-lab-home.sh" create "$LAB" >/dev/null
cat > "$LAB/.tasks.toml" <<'TOML'
backend = "markdown"

[markdown]
path = "data/backlog.md"
archive = "data/done-archive.md"
done_keep = 10
TOML
printf '%s\n' '## In flight' '' '## Queued' '' '## Done' > "$LAB/data/backlog.md"

mkdir -p "$W/basefull"
git -C "$WT" archive fb4463d9688432814dfc6c8d1f3ee594a0ce3ea9 | tar -x -C "$W/basefull"

gh repo clone "$REPO" "$W/clone" -- --quiet
cd "$W/clone"
git config user.name gsp-agent
git config user.email dale+gspagent@groovejones.com

make_pr() {  # <slug> -> prints PR number
  local slug=$1 branch="$RUN-$1" url
  git checkout -q -B "$branch" origin/main
  printf '%s %s\n' "$RUN" "$slug" > "$RUN-$slug.txt"
  git add "$RUN-$slug.txt"
  git commit -qm "live hold-label check: $slug"
  git push -q -u origin "$branch" 2>/dev/null
  url=$(gh pr create -R "$REPO" --base main --head "$branch" \
    --title "nm live check $RUN $slug" --body "Disposable PR for the fm-pr-merge hold-label live check ($RUN).")
  printf '%s\n' "${url##*/}"
}

A=$(make_pr base-hold)
B=$(make_pr head-hold)
C=$(make_pr head-relabel-paged)
D=$(make_pr head-control)

gh pr edit "$A" -R "$REPO" --add-label hold >/dev/null
gh pr edit "$B" -R "$REPO" --add-label hold >/dev/null
gh pr edit "$D" -R "$REPO" --add-label enhancement >/dev/null
# C: hold added then removed on the first timeline page, then 110 comments push
# the history past GitHub's 100-item page, then hold is added again.
gh pr edit "$C" -R "$REPO" --add-label hold >/dev/null
sleep 2
gh pr edit "$C" -R "$REPO" --remove-label hold >/dev/null
for i in $(seq 110); do
  gh api -X POST "repos/$REPO/issues/$C/comments" -f body="filler $i" >/dev/null
done
sleep 2
gh pr edit "$C" -R "$REPO" --add-label hold >/dev/null

for t in r2base r2hold r2holdun r2paged r2case r2control; do
  {
    printf 'window=fm-%s\n' "$t"
    printf 'worktree=%s\n' "$W/clone"
    printf 'project=%s\n' "$W/clone"
    printf 'kind=ship\nmode=no-mistakes\n'
  } > "$LAB/state/$t.meta"
done

cat > "$W/env" <<ENV
RUN=$RUN
REPO=$REPO
LAB=$LAB
PR_A=$A
PR_B=$B
PR_C=$C
PR_D=$D
ENV
cat "$W/env"

===== scripts/gh-run.sh =====
#!/usr/bin/env bash
# Run one live GitHub scenario: the base or the change's bin/fm-pr-merge.sh
# against a real PR in the private scratch repo, with the PR's live state and
# label history read before and after.
# Usage: gh-run.sh <evidence-name> <base|head> <task-id> <pr-number> [merge args...]
set -uo pipefail
WT=/Users/dalecarman/.agent-scratch/nm-runs/a5e67f4ea77f/01M5Z5SSDSR36DJ1JGJNQZT1KG
W=/var/folders/2j/3tf17x5d7pj10gdjbp1ydpf40000gn/T/nm-hold-r2.5ycrT3
EV=/Users/dalecarman/.no-mistakes/evidence/01M5Z5SSDSR36DJ1JGJNQZT1KG
. "$W/env"
name=$1 which=$2 task=$3 pr=$4
shift 4
case "$which" in
  base) SCRIPT="$W/basefull/bin/fm-pr-merge.sh"; rev="base fb4463d9" ;;
  head) SCRIPT="$WT/bin/fm-pr-merge.sh"; rev="change $(git -C "$WT" rev-parse --short=8 HEAD)" ;;
  *) echo "which must be base or head" >&2; exit 2 ;;
esac
URL="https://github.com/$REPO/pull/$pr"
export GH_TOKEN
GH_TOKEN=$(gh auth token --user gsp-agent)

pr_state() {
  gh pr view "$URL" --json number,state,mergedAt,mergeCommit,labels \
    --jq '"#\(.number) state=\(.state) mergedAt=\(.mergedAt // "none") mergeCommit=\(.mergeCommit.oid // "none") labels=\([.labels[].name] | join(","))"'
}

{
  echo "## $name"
  echo "date (UTC): $(date -u +%Y-%m-%dT%H:%M:%SZ)"
  echo "script: bin/fm-pr-merge.sh at $rev"
  echo "command: FM_HOME=<lab> bin/fm-pr-merge.sh $task $URL $*"
  echo "--- PR before:"
  pr_state
  echo "--- label history on the PR (GitHub timeline, every page; position = index in the timeline):"
  gh api --paginate "repos/$REPO/issues/$pr/timeline?per_page=100" --jq '.[] | .event' \
    | awk '{n++} END {print "timeline events: " n}'
  gh api --paginate "repos/$REPO/issues/$pr/timeline?per_page=100" --jq '.[] | .event + " " + (.label.name // "") + " " + (.actor.login // "") + " " + (.created_at // "")' \
    | awk '{n++} $1 == "labeled" || $1 == "unlabeled" {printf "  #%d (page %d) %s %s by %s at %s\n", n, int((n-1)/100)+1, $1, $2, $3, $4}'
  echo "--- fm-pr-merge.sh output (stdout+stderr):"
  (cd "$WT" && env -u NO_MISTAKES_GATE -u FM_GATE_REFUSE_BYPASS -u FM_ROOT_OVERRIDE -u FM_STATE_OVERRIDE -u FM_DATA_OVERRIDE \
     -u FM_CONFIG_OVERRIDE -u FM_PROJECTS_OVERRIDE FM_HOME="$LAB" \
     "$SCRIPT" "$task" "$URL" "$@" 2>&1)
  echo "--- exit status: $?"
  echo "--- PR after:"
  pr_state
} 2>&1 | no-mistakes redact > "$EV/$name.txt"
cat "$EV/$name.txt"

  -e FORGEJO__security__INSTALL_LOCK=true \
  -e FORGEJO__database__DB_TYPE=sqlite3 \
  -e "FORGEJO__server__ROOT_URL=$BASE_URL/" \
  -e FORGEJO__server__HTTP_PORT=3000 \
  -e FORGEJO__service__DISABLE_REGISTRATION=true \

printf '%s\n' "$C" > "$W/fj-container"
for _ in $(seq 120); do
  curl -fsS "$API/version" >/dev/null 2>&1 && break
  sleep 1

fj_user() {  # <name> <admin:true|false>
  local extra=()
  [ "$2" = true ] && extra=(--admin)
  docker exec -u git "$C" forgejo admin user create "${extra[@]}" --username "$1" \
    --email "$1@example.invalid" --random-password --must-change-password=false >/dev/null
  ( umask 077
    printf 'Authorization: [REDACTED]' \
      "$(docker exec -u git "$C" forgejo admin user generate-access-token [REDACTED] "$1" \
          --token-name nm-lab --scopes all --raw)" > "$FJ/$1.hdr" )
}
fj_user fixture true
fj_user cutover-owner false

# api <user> <method> <path> [json]
api() {
  local user=$1 method=$2 path=$3 body=${4:-}
  if [ -n "$body" ]; then
    curl -fsS -X "$method" -H @"$FJ/$user.hdr" -H 'Content-Type: application/json' \
      -d "$body" "$API$path"
  else
    curl -fsS -X "$method" -H @"$FJ/$user.hdr" "$API$path"
  fi
}

api fixture POST /user/repos '{"name":"fixture-repo","auto_init":true,"default_branch":"main","private":false}' >/dev/null
api fixture PUT /repos/fixture/fixture-repo/collaborators/cutover-owner '{"permission":"write"}' >/dev/null
HOLD_ID=$(api fixture POST /repos/fixture/fixture-repo/labels '{"name":"hold","color":"#ff0000"}' | jq -r .id)
READY_ID=$(api fixture POST /repos/fixture/fixture-repo/labels '{"name":"ready","color":"#00aa00"}' | jq -r .id)

make_pr() {  # <slug> -> PR number; commit on a new branch, green status at its head
  local slug=$1 content number sha
  content=$(printf '%s\n' "$slug" | base64)
  api fixture POST "/repos/fixture/fixture-repo/contents/$slug.txt" \
    "{\"branch\":\"main\",\"new_branch\":\"$slug\",\"content\":\"$content\",\"message\":\"live check $slug\"}" >/dev/null
  number=$(api fixture POST /repos/fixture/fixture-repo/pulls \
    "{\"head\":\"$slug\",\"base\":\"main\",\"title\":\"live check $slug\"}" | jq -r .number)
  sha=$(api fixture GET "/repos/fixture/fixture-repo/pulls/$number" | jq -r .head.sha)
  api fixture POST "/repos/fixture/fixture-repo/statuses/$sha" \
    '{"state":"success","context":"ci/lab","description":"lab"}' >/dev/null
  printf '%s\n' "$number"
}
label_add() { api "$1" POST "/repos/fixture/fixture-repo/issues/$2/labels" "{\"labels\":[$3]}" >/dev/null; }
label_del() { api "$1" DELETE "/repos/fixture/fixture-repo/issues/$2/labels/$3" >/dev/null; }

F1=$(make_pr fj-base-hold)
F2=$(make_pr fj-head-hold)
F3=$(make_pr fj-head-relabel-paged)
F4=$(make_pr fj-head-control)

label_add cutover-owner "$F1" "$HOLD_ID"
label_add cutover-owner "$F2" "$HOLD_ID"
label_add fixture "$F4" "$READY_ID"
# F3: fixture adds hold, fixture removes it, sixty comments push the history
# past Forgejo's 50-item page, then cutover-owner adds hold again.
label_add fixture "$F3" "$HOLD_ID"
sleep 2
label_del fixture "$F3" "$HOLD_ID"
for i in $(seq 60); do
  api fixture POST "/repos/fixture/fixture-repo/issues/$F3/comments" "{\"body\":\"filler $i\"}" >/dev/null
done
sleep 2
label_add cutover-owner "$F3" "$HOLD_ID"

# Isolated tea login for the merge script, pointing at the local instance.
TOKEN=$(sed 's/^Authorization: [REDACTED]' "$FJ/fixture.hdr")
( umask 077
  cat > "$W/fjhome/.config/tea/config.yml" <<YML
logins:
- name: fm-lab-fj
  url: $BASE_URL
  token: [REDACTED]
  default: true
  ssh_host: localhost
  insecure: false
  user: fixture
preferences:
  editor: false
  flag_defaults:
    remote: ""
YML
)
unset TOKEN

# Task worktree: a clone of the fixture repo.
git clone -q "$BASE_URL/fixture/fixture-repo.git" "$W/fjclone"
for t in fjbase fjhold fjrelabel fjcontrol; do
  {
    printf 'window=fm-%s\n' "$t"
    printf 'worktree=%s\n' "$W/fjclone"
    printf 'project=%s\n' "$W/fjclone"
    printf 'kind=ship\nmode=no-mistakes\n'

done

#!/usr/bin/env bash
# Stand up a disposable GitLab CE container on localhost, two users, one
# project, four MRs each with a successful external pipeline at its head, an
# isolated glab config, and lab metas for each GitLab scenario.
# Usage: gl-setup.sh start|seed
set -euo pipefail
W=/var/folders/2j/3tf17x5d7pj10gdjbp1ydpf40000gn/T/nm-hold-r2.5ycrT3
. "$W/env"
PORT=8929
BASE_URL="http://localhost:$PORT"
API="$BASE_URL/api/v4"
C="nm-gl-hold-$RUN"
GL="$W/gl"

case "${1:-}" in
start)
  docker run -d --name "$C" -p "127.0.0.1:$PORT:$PORT" --shm-size 256m \
    -e GITLAB_OMNIBUS_CONFIG="external_url '$BASE_URL'; gitlab_rails['gitlab_shell_ssh_port'] = 2224; puma['worker_processes'] = 0; sidekiq['concurrency'] = 5; prometheus_monitoring['enable'] = false; gitlab_kas['enable'] = false; registry['enable'] = false; gitlab_rails['usage_ping_enabled'] = false" \
    gitlab/gitlab-ce:latest >/dev/null
  printf '%s\n' "$C" > "$W/gl-container"
  echo "started $C"
  exit 0
  ;;
seed) ;;
*) echo "usage: gl-setup.sh start|seed" >&2; exit 2 ;;
esac

docker cp "$W/scripts/gl-tokens.rb" "$C:/tmp/nm-tokens.rb"
docker exec "$C" gitlab-rails runner /tmp/nm-tokens.rb
( umask 077
  docker exec "$C" cat /tmp/nm-tokens > "$GL/tokens"
  awk '$1 == "root" {print "PRIVATE-TOKEN: [REDACTED]$GL/tokens" > "$GL/root.hdr"
  awk '$1 == "cutover-owner" {print "PRIVATE-TOKEN: [REDACTED]$GL/tokens" > "$GL/cutover-owner.hdr" )
docker exec "$C" rm -f /tmp/nm-tokens

# api <user> <method> <path> [json]
api() {
  local user=$1 method=$2 path=$3 body=${4:-}
  if [ -n "$body" ]; then
    curl -fsS -X "$method" -H @"$GL/$user.hdr" -H 'Content-Type: application/json' \
      -d "$body" "$API$path"
  else
    curl -fsS -X "$method" -H @"$GL/$user.hdr" "$API$path"
  fi
}

PID=$(api root POST /projects '{"name":"fixture-repo","path":"fixture-repo","initialize_with_readme":true,"default_branch":"main","visibility":"private"}' | jq -r .id)
OWNER_ID=$(api root GET '/users?username=cutover-owner' | jq -r '.[0].id')
api root POST "/projects/$PID/members" "{\"user_id\":$OWNER_ID,\"access_level\":40}" >/dev/null
api root POST "/projects/$PID/labels" '{"name":"hold","color":"#ff0000"}' >/dev/null
api root POST "/projects/$PID/labels" '{"name":"ready","color":"#00aa00"}' >/dev/null

make_mr() {  # <slug> -> MR iid; commit on a new branch, successful external pipeline at its head
  local slug=$1 iid sha
  api root POST "/projects/$PID/repository/commits" \
    "{\"branch\":\"$slug\",\"start_branch\":\"main\",\"commit_message\":\"live check $slug\",\"actions\":[{\"action\":\"create\",\"file_path\":\"$slug.txt\",\"content\":\"$slug\\n\"}]}" >/dev/null
  iid=$(api root POST "/projects/$PID/merge_requests" \
    "{\"source_branch\":\"$slug\",\"target_branch\":\"main\",\"title\":\"live check $slug\"}" | jq -r .iid)
  sha=$(api root GET "/projects/$PID/repository/branches/$slug" | jq -r .commit.id)
  api root POST "/projects/$PID/statuses/$sha" \
    "{\"state\":\"success\",\"ref\":\"$slug\",\"name\":\"ci/lab\",\"description\":\"lab\"}" >/dev/null
  printf '%s\n' "$iid"

# Runs inside the disposable GitLab container: create the cutover-owner user,

# only root in the container can read.

  attrs = { username: 'cutover-owner', name: 'Cutover Owner', email: 'cutover-owner@example.invalid',
            password: [REDACTED] + 'Aa1!', skip_confirmation: true }
  owner = Users::CreateService.new(User.find_by_username('root'), attrs.merge(organization_id: org&.id).compact).execute
  owner = owner.payload[:user] if owner.respond_to?(:payload)
  owner.confirm if owner.respond_to?(:confirm) && !owner.confirmed?

end

#!/usr/bin/env bash
# Run one live GitLab scenario: base or the change's bin/fm-pr-merge.sh against
# an MR on the disposable local GitLab CE, reading the MR's live state and its
# label events before and after. Real glab with an isolated config only.
# Usage: gl-run.sh <evidence-name> <base|head> <task-id> <mr-iid> [merge args...]
set -uo pipefail
WT=/Users/dalecarman/.agent-scratch/nm-runs/a5e67f4ea77f/01M5Z5SSDSR36DJ1JGJNQZT1KG
W=/var/folders/2j/3tf17x5d7pj10gdjbp1ydpf40000gn/T/nm-hold-r2.5ycrT3
EV=/Users/dalecarman/.no-mistakes/evidence/01M5Z5SSDSR36DJ1JGJNQZT1KG
. "$W/env"
name=$1 which=$2 task=$3 mr=$4
shift 4
case "$which" in
  base) SCRIPT="$W/basefull/bin/fm-pr-merge.sh"; rev="base fb4463d9" ;;
  head) SCRIPT="$WT/bin/fm-pr-merge.sh"; rev="change $(git -C "$WT" rev-parse --short=8 HEAD)" ;;
  *) echo "which must be base or head" >&2; exit 2 ;;
esac
API="http://localhost:$GL_PORT/api/v4/projects/$GL_PID"
URL="https://localhost/root/fixture-repo/-/merge_requests/$mr"

mr_state() {
  curl -fsS -H @"$W/gl/root.hdr" "$API/merge_requests/$mr" \
    | jq -r '"!\(.iid) state=\(.state) merged_at=\(.merged_at // "none") merge_commit=\(.merge_commit_sha // "none") labels=\(.labels | join(","))"'
}

{
  echo "## $name"
  echo "date (UTC): $(date -u +%Y-%m-%dT%H:%M:%SZ)"
  echo "forge: disposable GitLab CE $(curl -fsS -H @"$W/gl/root.hdr" "http://localhost:$GL_PORT/api/v4/version" | jq -r .version) on localhost:$GL_PORT (real $(glab --version | head -1 | awk '{print $1, $2}'), isolated GLAB_CONFIG_DIR mapping host localhost to it)"
  echo "script: bin/fm-pr-merge.sh at $rev"
  echo "command: FM_HOME=<lab> bin/fm-pr-merge.sh $task $URL $*"
  echo "--- MR before:"
  mr_state
  echo "--- label history on the MR (GitLab resource label events):"
  curl -fsS -H @"$W/gl/root.hdr" "$API/merge_requests/$mr/resource_label_events?per_page=100" \
    | jq -r '.[] | "  \(.action) \(.label.name) by \(.user.username) at \(.created_at)"'
  echo "--- fm-pr-merge.sh output (stdout+stderr):"
  (cd "$WT" && env -u NO_MISTAKES_GATE -u FM_GATE_REFUSE_BYPASS -u FM_ROOT_OVERRIDE -u FM_STATE_OVERRIDE -u FM_DATA_OVERRIDE \
     -u FM_CONFIG_OVERRIDE -u FM_PROJECTS_OVERRIDE -u GITLAB_TOKEN -u GITLAB_HOST FM_HOME="$LAB" \
     HOME="$W/glhome" GLAB_CONFIG_DIR="$W/glconf" \
     "$SCRIPT" "$task" "$URL" "$@" 2>&1)
  echo "--- exit status: $?"
  echo "--- MR after:"
  mr_state
} 2>&1 | no-mistakes redact 2>/dev/null > "$EV/$name.txt"
cat "$EV/$name.txt"
