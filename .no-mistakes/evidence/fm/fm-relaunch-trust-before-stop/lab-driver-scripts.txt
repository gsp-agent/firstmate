# Every lab process starts from an empty environment: no inherited gate,
# fleet-path override, TMUX or Herdr variable, the lab's private tmux socket
# directory, a lab treehouse root, and the disposable Claude config dir.
lab_run() {
  env -i HOME="$HOME" USER="${USER:-$(id -un)}" LOGNAME="${USER:-$(id -un)}" PATH="$PATH" \
    SHELL=/bin/zsh TERM=xterm-256color LANG=en_US.UTF-8 \
    TMUX_TMPDIR="$TMUXD" TREEHOUSE_ROOT="$ROOT/treehouse" FM_BACKEND=tmux \
    DISABLE_AUTOUPDATER=1 CLAUDE_CONFIG_DIR="$CFG" FM_HOME="$LAB" "$@"
}
lab_tmux() { env -u TMUX TMUX_TMPDIR="$TMUXD" tmux "$@"; }

meta_get() { sed -n "s/^$2=//p" "$LAB/state/$1.meta" | tail -1; }
window_of() { meta_get "$1" window; }
pane_cmd() { lab_tmux display-message -p -t "$(window_of "$1")" '#{pane_current_command}'; }
pane_pid() { lab_tmux display-message -p -t "$(window_of "$1")" '#{pane_pid}'; }
# The agent is the pane terminal's foreground process that is not a shell
# (pane_current_command shows Claude Code as its version string).
agent_pid() {
  local tty
  tty=$(lab_tmux display-message -p -t "$(window_of "$1")" '#{pane_tty}')
  ps -t "${tty#/dev/}" -o pid=,stat=,comm= | awk '$2 ~ /\+/ && $3 !~ /(^|\/|-)zsh$/ { print $1; exit }'
}
screen() { lab_tmux capture-pane -p -t "$(window_of "$1")" | sed -e '/^[[:space:]]*$/d' | tail -"${2:-12}"; }
sum() { [ -e "$1" ] && shasum -a 256 "$1" | awk '{print $1}' || echo absent; }
proj_flags() {
  jq -c --arg p "$(cd -P "$PROJ" && pwd -P)" \
    '.projects[$p] // {} | {hasTrustDialogAccepted, hasClaudeMdExternalIncludesApproved, hasClaudeMdExternalIncludesWarningShown}' "$STORE"
}
# wait_screen <id> <text> <seconds>
wait_screen() {
  local i
  for ((i = 0; i < $3; i += 2)); do
    lab_tmux capture-pane -p -t "$(window_of "$1")" 2>/dev/null | grep -qF "$2" && return 0
    sleep 2
  done
  return 1
}

#!/usr/bin/env bash
# Build the lab and spawn two real Claude ships (A and B) on the lab's private
# tmux server through the product's own fm-spawn.sh.
set -u
. "$(dirname "$0")/lib.sh"

"$BIN/fm-lab-home.sh" create "$LAB" >/dev/null || { echo "lab home create failed"; exit 1; }
"$BIN/fm-lab-home.sh" tmux-dir "$LAB" > "$TMUXD_FILE" || { echo "tmux-dir failed"; exit 1; }
TMUXD=$(cat "$TMUXD_FILE")
echo "lab home: $LAB"
echo "tmux dir: $TMUXD"

mkdir -p "$CFG" "$ROOT/treehouse" "$ROOT/origins" "$ROOT/external"
printf '{"hasCompletedOnboarding":true,"theme":"dark"}\n' > "$STORE"
printf 'tmux\n' > "$LAB/config/backend"
printf 'claude\n' > "$LAB/config/crew-harness"
printf 'auto\n' > "$LAB/config/claude-permission-mode"

# A lab project whose CLAUDE.md imports a file outside the repository, so the
# real Claude worker asks the external-imports question.
printf 'Lab-only shared guidance for the notes project.\n' > "$ROOT/external/shared.md"
seed="$ROOT/origins/notes-seed"
mkdir -p "$seed/notes"
git init -q -b main "$seed"
printf '"""Lab notes library."""\n\nNOTES = []\n' > "$seed/notes/__init__.py"
printf '# notes\n\nLab project.\n\n@%s/external/shared.md\n' "$ROOT" > "$seed/CLAUDE.md"
git -C "$seed" add -A
git -C "$seed" -c user.name=lab -c user.email=lab@example.invalid commit -q -m "seed notes"
git clone -q --bare "$seed" "$ROOT/origins/notes.git"
git clone -q "$ROOT/origins/notes.git" "$PROJ"
printf '# Projects\n\n- notes [local-only +yolo] - tiny lab notes library\n' > "$LAB/data/projects.md"

lab_run tmux -f /dev/null new-session -d -s firstmate -n lab -x 220 -y 60 -c "$ROOT" || { echo "tmux start failed"; exit 1; }

for id in "$A_ID" "$B_ID"; do
  (cd "$LAB" && lab_run "$BIN/fm-brief.sh" "$id" notes --mode local-only) >/dev/null || { echo "brief $id failed"; exit 1; }
  TASK_TEXT="Lab ship $id for a live relaunch validation. Do nothing beyond reading this brief." \
  SPEC_TEXT="Nothing is in scope; stay idle." \
    python3 - "$LAB/data/$id/brief.md" <<'PY' || exit 1
import os, sys
p = sys.argv[1]
t = open(p, encoding="utf-8").read()
t = t.replace("{TASK}", os.environ["TASK_TEXT"], 1).replace("{FIRSTMATE_SPEC}", os.environ["SPEC_TEXT"], 1)
open(p, "w", encoding="utf-8").write(t)
PY
  (cd "$LAB" && lab_run "$BIN/fm-tasks-axi.sh" add "$id" "lab ship $id" --kind ship --repo notes) >/dev/null || { echo "backlog $id failed"; exit 1; }
  echo "== spawn $id"
  (cd "$LAB" && lab_run "$BIN/fm-spawn.sh" "$id" "$PROJ" --mode local-only --yolo on --harness claude --model sonnet --effort low) 2>&1 | tail -5
  echo "spawn $id rc=${PIPESTATUS[0]}"
done

#!/usr/bin/env bash
# Scenario: relaunch a running Claude ship onto claude while its project entry
# holds Claude's own recorded external-imports decline.
# Usage: relaunch-declined.sh <id> <bin-dir> <label>
# <bin-dir> is the product's bin/ (the change under test, or the base tree for
# the failing-first comparison).
set -u
. "$(dirname "$0")/lib.sh"
id=$1 bin=$2 label=$3
proof="$ROOT/proof-$label"
mkdir -p "$proof"
cp -p "$LAB/state/$id.meta" "$proof/meta-before"
cp -p "$LAB/data/$id/brief.md" "$proof/brief-before"
cp -p "$STORE" "$proof/store-before"
before_pane=$(pane_pid "$id") before_agent=$(agent_pid "$id") before_cmd=$(pane_cmd "$id")
echo "### $label: relaunch of $id onto claude with a declined project entry"
echo "product bin: $bin"
echo "project entry before: $(proj_flags)"
echo "before: window=$(window_of "$id") pane_command=$before_cmd pane_pid=$before_pane agent_pid=$before_agent"
echo "journal before: $([ -e "$LAB/state/$id.control-relaunch" ] && echo present || echo absent)"
echo
echo "\$ fm-control.sh $id relaunch --harness claude --note 'trust declined: relaunch check'"
out=$(cd "$LAB" && lab_run FM_CONTROL_EXIT_WAIT=60 FM_CONTROL_LAUNCH_WAIT=90 \
  "$bin/fm-control.sh" "$id" relaunch --harness claude --note "trust declined: relaunch check" 2>&1)
rc=$?
printf '%s\n' "$out"
echo "exit code: $rc"
echo
sleep 3
after_cmd=$(pane_cmd "$id") after_pane=$(pane_pid "$id") after_agent=$(agent_pid "$id")
echo "after: window=$(window_of "$id") pane_command=$after_cmd pane_pid=$after_pane agent_pid=${after_agent:-none}"
if [ -n "$after_agent" ] && [ "$after_agent" = "$before_agent" ]; then
  echo "old agent: STILL RUNNING (same claude pid $before_agent)"
elif [ -z "$after_agent" ]; then
  echo "old agent: STOPPED, and no agent is running in the pane"
else
  echo "old agent: replaced (claude pid $before_agent -> $after_agent)"
fi
cmp -s "$proof/meta-before" "$LAB/state/$id.meta" && echo "task record: unchanged" || echo "task record: CHANGED"
cmp -s "$proof/brief-before" "$LAB/data/$id/brief.md" && echo "instructions: unchanged" || echo "instructions: CHANGED"
cmp -s "$proof/store-before" "$STORE" && echo "trust store: byte-identical" || echo "trust store: CHANGED"
echo "journal after: $([ -e "$LAB/state/$id.control-relaunch" ] && echo present || echo absent)"
echo "project entry after: $(proj_flags)"
echo "pane tail:"
screen "$id" 6

#!/usr/bin/env bash
# Scenario: bin/fm-claude-trust.sh --check is read-only and answers exactly
# what a registration would, on the lab's real worktree and project.
set -u
. "$(dirname "$0")/lib.sh"
wt=$(cd -P "$(meta_get "$A_ID" worktree)" && pwd -P)
p=$(cd -P "$PROJ" && pwd -P)
run_check() {  # <config-dir> <args...>
  local cfg=$1 out rc
  shift
  out=$(cd "$LAB" && lab_run CLAUDE_CONFIG_DIR="$cfg" "$BIN/fm-claude-trust.sh" --check "$@" 2>&1)
  rc=$?
  printf '%s\n' "$out"
  echo "exit code: $rc"
}

# 1. A declined copy of the lab store (the shape real Claude wrote above).
dcfg="$ROOT/check-declined"
mkdir -p "$dcfg"
jq --arg p "$p" '.projects[$p].hasClaudeMdExternalIncludesApproved = false | .projects[$p].hasClaudeMdExternalIncludesWarningShown = true' "$STORE" > "$dcfg/.claude.json"
before=$(sum "$dcfg/.claude.json")
echo "### --check <worktree> <project> against a declined project entry"
echo "project entry: $(jq -c --arg p "$p" '.projects[$p] | {hasClaudeMdExternalIncludesApproved, hasClaudeMdExternalIncludesWarningShown}' "$dcfg/.claude.json")"
run_check "$dcfg" "$wt" "$PROJ"
[ "$(sum "$dcfg/.claude.json")" = "$before" ] && echo "store: byte-identical" || echo "store: CHANGED"
ls -A "$dcfg" | grep -v '^\.claude\.json$' | grep -q . && echo "config dir: gained files" || echo "config dir: no new files"
echo

echo "### --check --project <project> against the same declined entry (fresh-slot form)"
run_check "$dcfg" --project "$PROJ"
[ "$(sum "$dcfg/.claude.json")" = "$before" ] && echo "store: byte-identical" || echo "store: CHANGED"
echo

# 2. The live lab store, whose project entry is approved.
before=$(sum "$STORE")
echo "### --check <worktree> <project> against the approved lab store"
run_check "$CFG" "$wt" "$PROJ"
[ "$(sum "$STORE")" = "$before" ] && echo "store: byte-identical" || echo "store: CHANGED"
echo

# 3. A Claude config dir that does not exist yet.
ncfg="$ROOT/check-missing/cfg"
echo "### --check <worktree> <project> with a config dir that does not exist"
run_check "$ncfg" "$wt" "$PROJ"
[ -e "$ncfg" ] && echo "config dir: CREATED" || echo "config dir: still absent"
echo

# 4. The registration (no --check) against the declined copy still refuses,
# so the check and the write path agree.
echo "### registration (no --check) against the declined copy"
out=$(cd "$LAB" && lab_run CLAUDE_CONFIG_DIR="$dcfg" "$BIN/fm-claude-trust.sh" "$wt" "$PROJ" 2>&1); rc=$?
printf '%s\n' "$out"
echo "exit code: $rc"

#!/usr/bin/env bash
# Answer the imports question in a Herdr lab ship's real Claude pane.
# Usage: hdecline.sh <no|yes> <id>...
set -u
. "$(dirname "$0")/hlib.sh"
answer=$1
shift
for id in "$@"; do
  hwait "$id" "Allow external CLAUDE.md file imports?" 120 || { echo "$id: dialog not on screen"; hscreen "$id" 8; continue; }
  if [ "$answer" = yes ]; then
    hrun pane send-keys "$(hpane "$id")" down >/dev/null
    sleep 1
    hrun pane read "$(hpane "$id")" --source visible | grep -q '❯ Yes, allow external imports' || { echo "$id: Yes row not selected"; continue; }
  else
    hrun pane read "$(hpane "$id")" --source visible | grep -q '❯ No, disable external imports' || { echo "$id: No row not selected"; continue; }
  fi
  hrun pane send-keys "$(hpane "$id")" enter >/dev/null
  echo "$id: answered $answer in Claude's own dialog"
done
sleep 10
echo "old clone entry:  $(hflags "$OLD")"
echo "live clone entry: $(hflags "$LIVE")"

#!/usr/bin/env bash
# Relaunch an old-clone Herdr ship onto claude and report what happened to its
# agent, record, and trust store.
# Usage: hrelaunch.sh <id> <bin-dir> <label>
set -u
. "$(dirname "$0")/hlib.sh"
id=$1 bin=$2 label=$3
proof="$ROOT/hproof-$label"
mkdir -p "$proof"
claude_pid() {  # <pane>
  hrun pane process-info --pane "$1" 2>/dev/null |
    jq -r '[.result.process_info.foreground_processes[]? | select(.argv0 == "claude" or ((.argv[0] // "") | endswith("/claude")))][0].pid // empty'
}
pane_agent() { hrun pane get "$1" 2>/dev/null | jq -r '.result.pane | "\(.agent // "none")/\(.agent_status // "none")"'; }
cp -p "$HLAB/state/$id.meta" "$proof/meta-before"
cp -p "$HSTORE" "$proof/store-before"
pane=$(hpane "$id")

#!/usr/bin/env bash
# Read a Herdr lab ship's current endpoint: Herdr's agent state and the
# Claude pid in its pane (and optionally an earlier pane).
# Usage: hstate.sh <id> [<earlier-pane>]
set -u
. "$(dirname "$0")/hlib.sh"
claude_pid() {
  hrun pane process-info --pane "$1" 2>/dev/null |
    jq -r '[.result.process_info.foreground_processes[]? | select(.argv0 == "claude" or ((.argv[0] // "") | endswith("/claude")))][0].pid // empty'
}
pane_agent() { hrun pane get "$1" 2>/dev/null | jq -r '.result.pane | "\(.agent // "none")/\(.agent_status // "none")"'; }
id=$1
p=$(hpane "$id")
echo "recorded pane $p: herdr agent=$(pane_agent "$p") claude_pid=$(claude_pid "$p")"
if [ -n "${2:-}" ]; then
  echo "earlier pane $2: herdr agent=$(pane_agent "$2" || true) claude_pid=$(claude_pid "$2")"
fi
